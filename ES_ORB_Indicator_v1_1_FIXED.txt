//@version=6
indicator("ES/MES ORB Retest Indicator v1.1 FIXED", overlay=true, max_labels_count=100, max_lines_count=100)

grpTime = "Time Settings"
i_timezone   = input.string("America/New_York", "Timezone", options=["America/New_York", "America/Chicago", "UTC"], group=grpTime)
i_orbStart   = input.string("0800", "ORB Start (HHMM)", group=grpTime)
i_orbEnd     = input.string("0815", "ORB End (HHMM)", group=grpTime)
i_tradeStart = input.string("0930", "Trade Start (HHMM)", group=grpTime)
i_tradeEnd   = input.string("1100", "Trade End (HHMM)", group=grpTime)

grpEntry = "Entry Settings"
i_bodyMinPct    = input.float(0.35, "Min Body % of ORB Range", minval=0.0, maxval=2.0, step=0.05, group=grpEntry)
i_bodyMinAtr    = input.float(0.75, "Min Body × ATR", minval=0.0, maxval=5.0, step=0.05, group=grpEntry)
i_atrLen        = input.int(14, "ATR Length", minval=1, group=grpEntry)
i_retestMode    = input.string("Midpoint", "Retest Mode", options=["Breakout", "Boundary", "Midpoint"], group=grpEntry)
i_retestTimeout = input.int(60, "Retest Timeout Bars", minval=1, maxval=300, group=grpEntry)

grpFilters = "Filters"
i_useVWAP  = input.bool(true, "Use VWAP Alignment", group=grpFilters)
i_useEMA   = input.bool(false, "Use EMA Alignment", group=grpFilters)
i_emaLen   = input.int(50, "EMA Length", minval=1, group=grpFilters)
i_minRange = input.float(5.0, "Min ORB Range Points", minval=0.25, step=0.25, group=grpFilters)
i_maxRange = input.float(50.0, "Max ORB Range Points", minval=1.0, step=0.5, group=grpFilters)

grpDisplay = "Display"
i_showDashboard = input.bool(true, "Show Dashboard", group=grpDisplay)
i_showSignals = input.bool(true, "Show Signals", group=grpDisplay)

f_timeToMins(string hhmm) =>
    float h = str.tonumber(str.substring(hhmm, 0, 2))
    float m = str.tonumber(str.substring(hhmm, 2, 4))
    int(h * 60 + m)

f_inWindow(string startHHMM, string endHHMM) =>
    int curMins = hour(time, i_timezone) * 60 + minute(time, i_timezone)
    int startMins = f_timeToMins(startHHMM)
    int endMins = f_timeToMins(endHHMM)
    startMins < endMins ? (curMins >= startMins and curMins < endMins) : (curMins >= startMins or curMins < endMins)

newDay = ta.change(dayofmonth(time, i_timezone)) != 0
inOrbWindow = f_inWindow(i_orbStart, i_orbEnd)
inTradeWindow = f_inWindow(i_tradeStart, i_tradeEnd)

var float orbHigh = na
var float orbLow = na
var float orbMid = na
var bool orbDone = false
var bool signalToday = false

if newDay
    orbHigh := na
    orbLow := na
    orbMid := na
    orbDone := false
    signalToday := false

if inOrbWindow and not orbDone
    orbHigh := na(orbHigh) ? high : math.max(orbHigh, high)
    orbLow := na(orbLow) ? low : math.min(orbLow, low)

if not inOrbWindow and not orbDone and not na(orbHigh) and not na(orbLow)
    orbDone := true
    orbMid := (orbHigh + orbLow) / 2.0

orbRange = orbDone ? orbHigh - orbLow : na
orbRangeOk = orbDone and orbRange >= i_minRange and orbRange <= i_maxRange

atrVal = ta.atr(i_atrLen)
barBody = math.abs(close - open)
bodyPctOfRange = orbRangeOk ? barBody / orbRange : 0.0
bodyVsAtr = atrVal > 0 ? barBody / atrVal : 0.0
displacementOk = bodyPctOfRange >= i_bodyMinPct or bodyVsAtr >= i_bodyMinAtr

vwapVal = ta.vwap(hlc3)
emaVal = ta.ema(close, i_emaLen)
longContextOk = (not i_useVWAP or close > vwapVal) and (not i_useEMA or close > emaVal)
shortContextOk = (not i_useVWAP or close < vwapVal) and (not i_useEMA or close < emaVal)

var int setupDir = 0
var int setupBar = na
var bool retestSeen = false

if newDay
    setupDir := 0
    setupBar := na
    retestSeen := false

canSeek = orbRangeOk and inTradeWindow and not signalToday

if canSeek and setupDir == 0 and displacementOk
    if close > orbHigh and longContextOk
        setupDir := 1
        setupBar := bar_index
        retestSeen := i_retestMode == "Breakout"
    else if close < orbLow and shortContextOk
        setupDir := -1
        setupBar := bar_index
        retestSeen := i_retestMode == "Breakout"

if setupDir != 0 and not na(setupBar) and bar_index - setupBar > i_retestTimeout
    setupDir := 0
    setupBar := na
    retestSeen := false

longRetestLevel = i_retestMode == "Midpoint" ? orbMid : orbHigh
shortRetestLevel = i_retestMode == "Midpoint" ? orbMid : orbLow

if setupDir == 1 and not retestSeen and low <= longRetestLevel
    retestSeen := true
if setupDir == -1 and not retestSeen and high >= shortRetestLevel
    retestSeen := true

longSig = canSeek and setupDir == 1 and retestSeen and close > orbHigh and displacementOk and longContextOk
shortSig = canSeek and setupDir == -1 and retestSeen and close < orbLow and displacementOk and shortContextOk

if longSig or shortSig
    signalToday := true
    setupDir := 0

plot(orbRangeOk ? orbHigh : na, "ORB High", color=color.new(color.green, 0), linewidth=2)
plot(orbRangeOk ? orbLow : na, "ORB Low", color=color.new(color.red, 0), linewidth=2)
plot(orbRangeOk ? orbMid : na, "ORB Mid", color=color.new(color.orange, 20), linewidth=1)
plot(i_useVWAP ? vwapVal : na, "VWAP", color=color.new(color.blue, 40), linewidth=1)
plot(i_useEMA ? emaVal : na, "EMA", color=color.new(color.aqua, 50), linewidth=1)

plotshape(i_showSignals and setupDir == 1 and bar_index == setupBar, title="Long Breakout", style=shape.triangleup, location=location.belowbar, color=color.green, size=size.tiny, text="BRK")
plotshape(i_showSignals and setupDir == -1 and bar_index == setupBar, title="Short Breakout", style=shape.triangledown, location=location.abovebar, color=color.red, size=size.tiny, text="BRK")
plotshape(i_showSignals and longSig, title="Long Ready", style=shape.triangleup, location=location.belowbar, color=color.lime, size=size.small, text="LONG")
plotshape(i_showSignals and shortSig, title="Short Ready", style=shape.triangledown, location=location.abovebar, color=color.orange, size=size.small, text="SHORT")

alertcondition(longSig, title="Long Setup Ready", message="ES/MES ORB LONG setup ready")
alertcondition(shortSig, title="Short Setup Ready", message="ES/MES ORB SHORT setup ready")

var table dash = table.new(position.top_right, 2, 8, border_width=1)
if i_showDashboard and barstate.islast
    table.cell(dash, 0, 0, "ES/MES ORB v1.1", text_color=color.white, bgcolor=color.new(color.blue, 70))
    table.cell(dash, 1, 0, "Status", text_color=color.white, bgcolor=color.new(color.blue, 70))
    table.cell(dash, 0, 1, "ORB")
    table.cell(dash, 1, 1, inOrbWindow ? "BUILDING" : orbDone ? "DONE" : "WAIT")
    table.cell(dash, 0, 2, "Range")
    table.cell(dash, 1, 2, orbDone ? str.tostring(orbRange, "#.##") : "na")
    table.cell(dash, 0, 3, "Range OK")
    table.cell(dash, 1, 3, orbRangeOk ? "YES" : "NO")
    table.cell(dash, 0, 4, "Trade Window")
    table.cell(dash, 1, 4, inTradeWindow ? "OPEN" : "CLOSED")
    table.cell(dash, 0, 5, "Setup")
    table.cell(dash, 1, 5, setupDir == 1 ? "LONG" : setupDir == -1 ? "SHORT" : "NONE")
    table.cell(dash, 0, 6, "Retest")
    table.cell(dash, 1, 6, retestSeen ? "SEEN" : "NO")
    table.cell(dash, 0, 7, "Signal Today")
    table.cell(dash, 1, 7, signalToday ? "YES" : "NO")
